cmake_minimum_required(VERSION 3.16)
project(restv_harness C CXX)
# PC harness: run N frames headless → frame and audio hashes, PNG dumps
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 14)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_C_FLAGS_RELEASE "-O2 -g")
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -g")
option(RSTV_DYNAREC "SH-2 recompiler (ARM builds only, arm-linux.cmake)" OFF)
if(RSTV_DYNAREC)
  enable_language(ASM)
endif()
include(${CMAKE_CURRENT_SOURCE_DIR}/../../src/core/core.cmake)
add_executable(harness main.c nbgcheck.c ../../vita/src/video/nbgq.c ../../vita/src/video/rbgq.c)
target_include_directories(harness PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../vita/src/video)
target_link_libraries(harness rstv_core z m pthread)
if(RSTV_DYNAREC)
  target_link_options(harness PRIVATE -static)
  target_compile_definitions(rstv_core PUBLIC RSTV_IOREPLAY=1)
endif()
target_compile_definitions(rstv_core PRIVATE RSTV_SH2_STATS=1 RSTV_VDP1_STATS=1)
# Run SCSP on the main thread for deterministic hashes (Vita uses an async thread)
target_compile_definitions(rstv_core PUBLIC RSTV_SYNC_SCSP=1)
target_compile_definitions(rstv_core PRIVATE RSTV_M68K_TRACE=1) # 68K instruction trace (RSTV_68KTRACE)

# System C / C2 harness (ADR-107)
add_executable(c2harness c2run.c)
target_link_libraries(c2harness rstv_c2 rstv_core z m pthread)
